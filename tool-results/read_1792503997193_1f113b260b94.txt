   305→/* گام‌های ۴ تا ۸ — شاخص‌ها */
   306→function ScoredStep({ stepKey, assessment }: { stepKey: string; assessment: Assessment }) {
   307→  const groups = groupsOfStep(stepKey);
   308→  const setAnswer = usePorsesh((s) => s.setAnswer);
   309→  const provider = assessment.members.find((m) => m.isMainProvider);
   310→
   311→  const membersFor = (g: QuestionGroup): Member[] => {
   312→    if (!g.loop) return [];
   313→    if (g.memberFilter === "schoolAge") {
   314→      const rel = assessment.members.filter((m) => m.schoolAge || (typeof m.age === "number" && m.age >= 6 && m.age <= 18));
   315→      return rel;
   316→    }
   317→    if (g.memberFilter === "provider") return provider ? [provider] : [];
   318→    return assessment.members;
   319→  };
   320→
   321→  return (
   322→    <div className="space-y-4">
   323→      {assessment.members.length === 0 && (
   324→        <div className="rounded-xl border border-amber-200 bg-amber-50 p-4 text-sm leading-6">
   325→          برای پاسخ به پرسش‌های این بخش، ابتدا در گام «اطلاعات پایه» اعضای خانوار را ثبت کنید.
   326→        </div>
   327→      )}
   328→
   329→      {groups.map((g) => {
   330→        const rel = membersFor(g);
   331→        return (
   332→          <div key={g.key} className="space-y-3">
   333→            {g.key === "D3" && <IncomeEditor assessment={assessment} />}
   334→            {g.key === "poverty" && <PovertyEditor assessment={assessment} />}
   335→            {g.key === "D1m" && !provider && assessment.members.length > 0 && (
   336→              <p className="text-sm rounded-xl border border-amber-200 bg-amber-50 p-3.5 leading-6">
