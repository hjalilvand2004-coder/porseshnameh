    11→
    12→export type QuestionType = "single" | "multi" | "number" | "text" | "textarea" | "boolean";
    13→
    14→export interface Option {
    15→  value: string;
    16→  label: string;
    17→  desc?: string;
    18→  /** پاسخ خاص — بدون امتیازدهی خودکار */
    19→  special?: boolean;
    20→}
    21→
    22→export interface Question {
    23→  id: string;
    24→  title: string;
    25→  help?: string;
    26→  type: QuestionType;
    27→  level: "household" | "member";
    28→  /** فیلتر عضو برای پرسش‌های سطح عضو */
    29→  memberFilter?: "all" | "schoolAge" | "provider";
    30→  indicator?: IndicatorId;
    31→  dimension?: DimensionId;
    32→  options?: Option[];
    33→  unit?: string;
    34→  min?: number;
    35→  max?: number;
    36→  /** شاخص‌های حساس — نمایش محدود در نمای خانوار */
    37→  sensitive?: boolean;
    38→  /** آیا در محاسبه اثر دارد (اطلاع‌رسانی برای شفافیت) */
    39→  scored?: boolean;
    40→  placeholder?: string;
    41→}
    42→
    43→export interface QuestionGroup {
    44→  /** کلید منطقی گروه */
    45→  key: string;
    46→  title: string;
    47→  indicator: IndicatorId;
    48→  /** گروه سطح عضو حلقه می‌خورد */
    49→  loop?: boolean;
    50→  memberFilter?: "all" | "schoolAge" | "provider";
    51→  questions: Question[];
    52→}
