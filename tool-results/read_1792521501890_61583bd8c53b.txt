    34→export interface ProgressInfo {
    35→  applicable: number;
    36→  answered: number;
    37→  pct: number;
    38→  /** درصد پاسخ هر بُعد A..E */
    39→  dims: Record<string, { pct: number; answered: number; applicable: number }>;
    40→}
    41→
    42→/** درصد تکمیل پاسخ‌های پرسشنامه — فقط پرسش‌های دارای پاسخ قابل‌شمارش (نه فیلدهای آزاد) */
    43→export function computeProgress(assessment: Assessment): ProgressInfo {
    44→  const provider = assessment.members.find((m) => m.isMainProvider);
    45→  let applicable = 0;
    46→  let answered = 0;
    47→  const dims: ProgressInfo["dims"] = {};
    48→  const bump = (d: string | undefined, ok: boolean) => {
    49→    if (!d) return;
    50→    const cur = dims[d] ?? { pct: 0, answered: 0, applicable: 0 };
    51→    cur.applicable += 1;
    52→    if (ok) cur.answered += 1;
    53→    cur.pct = cur.answered / cur.applicable;
    54→    dims[d] = cur;
    55→  };
    56→  for (const g of QUESTION_GROUPS) {
    57→    const list: (Member | null)[] = g.loop
    58→      ? (g.memberFilter === "schoolAge"
    59→          ? assessment.members.filter((m) => m.schoolAge || (typeof m.age === "number" && m.age >= 6 && m.age <= 18))
    60→          : g.memberFilter === "provider" ? (provider ? [provider] : []) : assessment.members)
    61→      : [null];
    62→    const qs = g.questions.filter((qq) => ["single", "multi", "boolean", "number"].includes(qq.type));
    63→    for (const m of list) {
    64→      for (const qq of qs) {
    65→        applicable += 1;
    66→        const key = m ? `${qq.id}#${m.id}` : qq.id;
    67→        const v = assessment.answers[key];
    68→        const has = v !== null && v !== undefined && v !== "" && !(Array.isArray(v) && v.length === 0);
    69→        if (has) answered += 1;
    70→        bump(qq.dimension, has);
    71→      }
    72→    }
    73→  }
    74→  return { applicable, answered, pct: applicable ? answered / applicable : 0, dims };
    75→}

    77→function DimensionChips({ dims }: { dims: ProgressInfo["dims"] }) {
    78→  const D: { id: string; label: string }[] = [
    79→    { id: "A", label: "سلامت" }, { id: "B", label: "آموزش" }, { id: "C", label: "مسکن" },
    80→    { id: "D", label: "معیشت" }, { id: "E", label: "مراقبت" },
    81→  ];
    82→  return (
    83→    <div className="flex flex-wrap items-center gap-1.5">
    84→      {D.map((d) => {
    85→        const st = dims[d.id];
    86→        const p = st ? Math.round(st.pct * 100) : 0;
    87→        return (
    88→          <span key={d.id} title={`تکمیل بُعد ${d.label}: ${faNum(p)}٪`} className="inline-flex items-center gap-1.5 rounded-full border bg-card px-2 py-0.5 text-[10px] whitespace-nowrap">
    89→            <b className="text-teal-strong">{d.id}</b> {d.label}
    90→            <span className="inline-block w-8 h-1 rounded-full bg-muted overflow-hidden align-middle">
    91→              <span className="block h-full bg-teal-strong rounded-full transition-all" style={{ width: `${p}%` }} />
    92→            </span>
    93→            {faNum(p)}٪
    94→          </span>
    95→        );
    96→      })}
    97→    </div>
    98→  );
    99→}

   327→  const postFull = ws.post?.status === "ok";

   329→  const field = (key: keyof Assessment["household"], label: string, opts?: { dir?: "ltr" | "rtl"; type?: "text" | "number"; placeholder?: string; locked?: string }) => {
